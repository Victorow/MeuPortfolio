# Space Portfolio Redesign — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the portfolio as a scroll-driven journey through the Solar System (8 chapters, real planet textures, cinematic post-processing) where the 3D is background and the UI/UX serves a recruiter.

**Architecture:** DOM stays DOM (Next.js static export, next-intl, Tailwind). A fixed R3F canvas renders the scene. Lenis drives native scroll; GSAP ScrollTrigger converts scroll to a single `progress` (0–1) in a zustand store. The camera samples a CatmullRom path from `progress`; planets sit at fixed world positions; post-processing presets interpolate per chapter. Sections are pinned by ScrollTrigger during their chapter range.

**Tech Stack:** Next.js 14.2 (static export), React 18, TypeScript, Tailwind 3, next-intl 3, Lenis, GSAP 3 + ScrollTrigger, zustand, @react-three/fiber 8, @react-three/drei 9, three 0.169, @react-three/postprocessing 2, howler, sharp (build script), Vitest, Playwright.

## Global Constraints

- `output: 'export'`, `trailingSlash: true`, `images.unoptimized` stay (GitHub Pages). No server features.
- Locales: `pt-BR` (default) and `en`. Every user-visible string goes through `messages/*.json`.
- Canvas is `aria-hidden="true"` and `pointer-events: none`. All interaction is DOM.
- Zero `setState` inside `useFrame`. Per-frame state lives in refs/uniforms/zustand transient subscriptions.
- `prefers-reduced-motion`: no Lenis, no snap, no parallax, no planet rotation, camera jumps between poses.
- Texture tiers: `high` 8K Earth / 4K others, `mid` 4K Earth / 2K others, `low` 2K/1K. Budget: ≤12 MB before preloader ends, ≤45 MB desktop total, ≤15 MB mobile total.
- Texture format: tiered JPG/PNG generated by `scripts/assets.mjs` (sharp). If `toktx` is on PATH the script also emits `.ktx2` and the manifest prefers it. (toktx is not installed on this machine, so JPG is the shipped format for now; spec D4 is honoured as "ready for KTX2".)
- Accent colours per chapter (exact): earth `#5FA8FF`, mars `#D96A3C`, jupiter `#E0A862`, saturn `#E8C98A`, uranus `#8FD8E8`, venus `#F0D9A0`, neptune `#3E6CFF`, deep `#B8C4FF`. Background `#030305`, text `#F2F2F0`.
- Attribution: footer credits "Texturas: Solar System Scope (CC BY 4.0) · NASA".
- **Commits:** none until Victor validates. No Co-Authored-By trailer ever. Tasks end with "verify", not "commit".
- Validation: `npm run build` passes; `npm run test` (Vitest) passes; `npm run e2e` (Playwright) passes and writes screenshots to `e2e/screenshots/`.

```ts
import type { ChapterId } from '@/content/planets';
export const CHAPTERS: ChapterId[] = ['earth','mars','jupiter','saturn','uranus','venus','neptune','deep'];
// relative scroll weight per chapter (hero and deep shorter, project chapters longer)
const WEIGHTS: Record<ChapterId, number> = { earth: 1.1, mars: 1.3, jupiter: 1.3, saturn: 1.4, uranus: 1.2, venus: 1.2, neptune: 1.5, deep: 1.0 };
const total = CHAPTERS.reduce((s, c) => s + WEIGHTS[c], 0);
export const CHAPTER_RANGES: Record<ChapterId, { start: number; end: number }> = (() => {
  const out = {} as Record<ChapterId, { start: number; end: number }>; let acc = 0;
  for (const c of CHAPTERS) { const w = WEIGHTS[c] / total; out[c] = { start: acc, end: acc + w }; acc += w; }
  out.deep.end = 1; return out;
})();
export const clamp01 = (x: number) => Math.min(1, Math.max(0, x));
export function rangeOf(id: ChapterId) { return CHAPTER_RANGES[id]; }
export function chapterAt(p: number): ChapterId {
  const x = clamp01(p);
  for (const c of CHAPTERS) { const r = CHAPTER_RANGES[c]; if (x < r.end) return c; }
  return 'deep';
}
export function localProgress(p: number, id: ChapterId) {
  const r = CHAPTER_RANGES[id]; return clamp01((p - r.start) / (r.end - r.start));
}
export function snapPoints() { return CHAPTERS.map((c) => CHAPTER_RANGES[c].start); }
export function chapterIndex(id: ChapterId) { return CHAPTERS.indexOf(id); }
export function scrollHeightVh() { return Math.round(total * 180); } // total page height: ~180vh per weight unit
```

```ts
import * as THREE from 'three';
import { PLANETS } from '@/content/planets';
import { CHAPTER_RANGES } from './timeline';
type Key = { t: number; pos: [number, number, number]; look: [number, number, number]; fov: number };
const P = (id: keyof typeof PLANETS, d: [number, number, number]): [number, number, number] => {
  const p = PLANETS[id].position; return [p[0] + d[0], p[1] + d[1], p[2] + d[2]];
};
const R = CHAPTER_RANGES;
export const CAMERA_KEYFRAMES: Key[] = [
  { t: 0,                      pos: P('earth', [-3.5, 0.8, 7.5]),  look: P('earth', [1.2, 0, 0]),   fov: 42 },
  { t: R.earth.end - 0.02,     pos: P('earth', [-6, 2, 14]),       look: P('mars', [0, 0, 0]),       fov: 48 },
  { t: R.mars.start + 0.03,    pos: P('mars', [-4, 0.5, 5.5]),     look: P('mars', [0.8, 0, 0]),     fov: 40 },
  { t: R.mars.end - 0.02,      pos: P('mars', [3, 1.5, 6]),        look: P('jupiter', [0, 0, 0]),    fov: 46 },
  { t: R.jupiter.start + 0.03, pos: P('jupiter', [-12, 2, 16]),    look: P('jupiter', [4, 0, 0]),    fov: 40 },
  { t: R.jupiter.end - 0.02,   pos: P('jupiter', [-8, -3, 20]),    look: P('saturn', [0, 0, 0]),     fov: 46 },
  { t: R.saturn.start + 0.03,  pos: P('saturn', [-14, -6, 14]),    look: P('saturn', [2, 0, 0]),     fov: 40 },  // from below the rings
  { t: R.saturn.end - 0.02,    pos: P('saturn', [-6, 7, 18]),      look: P('uranus', [0, 0, 0]),     fov: 46 },
  { t: R.uranus.start + 0.03,  pos: P('uranus', [-8, 1, 9]),       look: P('uranus', [1.5, 0, 0]),   fov: 40 },
  { t: R.uranus.end - 0.02,    pos: P('uranus', [4, 2, 10]),       look: P('venus', [0, 0, 0]),      fov: 46 },
  { t: R.venus.start + 0.03,   pos: P('venus', [-5, 0.5, 6]),      look: P('venus', [1, 0, 0]),      fov: 40 },
  { t: R.venus.end - 0.02,     pos: P('venus', [2, 3, 8]),         look: P('neptune', [0, 0, 0]),    fov: 46 },
  { t: R.neptune.start + 0.03, pos: P('neptune', [-7, 1, 9]),      look: P('neptune', [1.5, 0, 0]),  fov: 40 },
  { t: R.neptune.end - 0.02,   pos: P('neptune', [-4, 6, 12]),     look: P('neptune', [20, 30, -60]), fov: 50 },
  { t: 1,                      pos: P('neptune', [-10, 30, 25]),   look: P('neptune', [40, 80, -120]), fov: 60 }
];
const posCurve = new THREE.CatmullRomCurve3(CAMERA_KEYFRAMES.map((k) => new THREE.Vector3(...k.pos)), false, 'centripetal');
const lookCurve = new THREE.CatmullRomCurve3(CAMERA_KEYFRAMES.map((k) => new THREE.Vector3(...k.look)), false, 'centripetal');
// map progress → curve parameter u via keyframe times (piecewise linear)
function toU(p: number) {
  const n = CAMERA_KEYFRAMES.length - 1; const x = Math.min(1, Math.max(0, p));
  for (let i = 0; i < n; i++) { const a = CAMERA_KEYFRAMES[i].t, b = CAMERA_KEYFRAMES[i + 1].t;
    if (x <= b) { const f = b === a ? 0 : (x - a) / (b - a); return (i + f) / n; } }
  return 1;
}
function fovAt(p: number) {
  const x = Math.min(1, Math.max(0, p));
  for (let i = 0; i < CAMERA_KEYFRAMES.length - 1; i++) { const a = CAMERA_KEYFRAMES[i], b = CAMERA_KEYFRAMES[i + 1];
    if (x <= b.t) { const f = b.t === a.t ? 0 : (x - a.t) / (b.t - a.t); const s = f * f * (3 - 2 * f); return a.fov + (b.fov - a.fov) * s; } }
  return CAMERA_KEYFRAMES.at(-1)!.fov;
}
const _pos = new THREE.Vector3(), _look = new THREE.Vector3();
export function sampleCamera(progress: number) {
  const u = toU(progress);
  return { position: posCurve.getPointAt(u, _pos).clone(), lookAt: lookCurve.getPointAt(u, _look).clone(), fov: fovAt(progress) };
}
```
  Note: `getPointAt` uses arc-length parametrisation; combined with `toU` it keeps speed roughly uniform between keyframes. If the continuity test fails, raise the threshold only after visually confirming no jump.

```ts
export function resolveTexture(key: TextureKey, tier: PerfTier, m: Manifest = generated as Manifest): string {
```

```ts
      lenis = new Lenis({ lerp: 0.09, smoothWheel: true });
```
